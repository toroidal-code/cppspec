37inline std::string strip_type_keywords(std::string_view name) {
38 static constexpr std::string_view keywords[] = {"enum ", "class ", "struct ", "union "};
39 const auto continues_identifier = [](char c) {
40 return (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '_';
41 };
42 std::string stripped;
43 stripped.reserve(name.size());
44 std::size_t i = 0;
45 while (i < name.size()) {
46 const bool at_word_start = i == 0 || !continues_identifier(name[i - 1]);
47 const auto* keyword = std::ranges::find_if(
48 keywords, [&](std::string_view word) { return at_word_start && name.substr(i).starts_with(word); });
49 if (keyword != std::ranges::end(keywords)) {
50 i += keyword->size();
51 } else {
52 stripped += name[i];
53 ++i;
54 }
55 }
56 return stripped;
57}
