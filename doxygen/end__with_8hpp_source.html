22template <typename A, typename E>
23 requires Util::compares_start_and_end<A, E>
24class EndWith : public MatcherBase<A, E> {
25 static constexpr bool compares_text = Util::is_string_like<A> && Util::is_string_like<E>;
26
27 // Whether both values hold something to compare: neither is a null C string.
28 bool has_values() {
29 if constexpr (Util::is_string_like<A>) {
30 if (!string_view_of(this->actual())) {
31 return false;
32 }
33 }
34 if constexpr (Util::is_string_like<E>) {
35 return string_view_of(this->expected()).has_value();
36 } else {
37 return true;
38 }
39 }
40
41 // The actual value against the expected one, both known to be there.
42 bool compare() {
43 if constexpr (compares_text) {
44 return string_view_of(this->actual())->ends_with(*string_view_of(this->expected()));
45 } else {
46 A& actual = this->actual();
47 auto compare_with = [&actual](const auto& expected) {
48 // std::equal reads as many elements of the actual range as the expected one has, so a shorter actual range
49 // fails before it is read.
50 return std::ranges::distance(expected) <= std::ranges::distance(actual) &&
51 std::equal(std::ranges::rbegin(expected), std::ranges::rend(expected), std::ranges::rbegin(actual));
52 };
53 if constexpr (Util::is_string_like<E>) {
54 return compare_with(*string_view_of(this->expected()));
55 } else {
56 return compare_with(this->expected());
57 }
58 }
59 }
60
61 public:
62 EndWith(Expectation<A>& expectation, E start) : MatcherBase<A, E>(expectation, start) {}
63
64 std::string verb() override { return "end with"; }
65 std::string failure_message() override {
66 return MatcherBase<A, E>::failure_message() + no_text_to_search(this->actual()) +
67 no_text_to_look_for(this->expected());
68 }
69 std::string failure_message_when_negated() override {
70 return MatcherBase<A, E>::failure_message_when_negated() + no_text_to_search(this->actual()) +
71 no_text_to_look_for(this->expected());
72 }
73
74 bool match() override { return has_values() && compare(); }
75 bool negated_match() override { return has_values() && !compare(); }
76};
