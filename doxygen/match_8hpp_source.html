10// A std::regex keeps no copy of its source, so a pattern given as text is named by that text; one given as a
11// std::regex is named only as a regex.
12inline std::string describe_pattern(const std::string& pattern) {
13 return pattern.empty() ? std::string{"a regex"} : "\"" + pattern + "\"";
14}

20template <Util::is_string_like T>
21[[nodiscard]] std::regex regex_of(const T& pattern) {
22 const auto text = string_view_of(pattern);
23 return text ? std::regex(text->begin(), text->end()) : std::regex();

29template <Util::is_string_like T>
30[[nodiscard]] std::string pattern_name_of(const T& pattern) {
31 const auto text = string_view_of(pattern);
32 return text ? describe_pattern(std::string(*text)) : std::string("nullptr");
33}
