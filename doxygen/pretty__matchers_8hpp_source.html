31// C++26 (P3168) gives std::optional begin()/end(), so cxx-prettyprint's
32// has_begin_end/has_const_iterator detection now classifies optional<T> as a
33// container and tries to element-print its value_type — which hard-errors for
34// element types lacking operator<< (and is wrong regardless: an optional is not

215template <typename T>
216inline std::string Pretty::to_sentence(const std::vector<T>& objects) {
217 std::vector<std::string> words;
218 for (const auto& object : objects) {
219 words.push_back(to_word(object));
220 }
221
222 std::stringstream ss;
223 switch (words.size()) {
224 case 0:
225 return "";
226 case 1:
227 ss << " " << words[0];
228 break;
229 case 2:
230 ss << " " << words[0] << " and " << words[1];
231 break;
232 default:
233 ss << " ";
234 for (size_t i = 0; i < words.size() - 1; ++i) {
235 if (i != 0) {
236 ss << ", ";
237 }
238 ss << words[i];
239 }
240 ss << ", and " << words.back();
241 break;
242 }
243
244 return ss.str();
245}

371template <typename T>
372 requires Util::prints_elements_through_pretty<T>
373inline std::string Pretty::to_word(const T& item) {
374 const auto& delimiters = pretty_print::delimiters<T, char>::values;
375 std::string word = delimiters.prefix != nullptr ? delimiters.prefix : "";
376 bool first = true;
377 auto append = [&](const auto& element) {
378 if (!first && delimiters.delimiter != nullptr) {
379 word += delimiters.delimiter;
380 }
381 word += to_word(element);
382 first = false;
383 };
384 if constexpr (std::ranges::input_range<const T>) {
385 for (const auto& element : item) {
386 // std::vector<bool> may hand out a proxy for each element, as libc++'s does; it is printed as the bool it holds.
387 if constexpr (std::same_as<std::remove_cv_t<std::ranges::range_value_t<const T>>, bool>) {
388 append(static_cast<bool>(element));
389 } else {
390 append(element);
391 }
392 }
393 } else {
394 std::apply([&](const auto&... elements) { (append(elements), ...); }, item);
395 }
396 if (delimiters.postfix != nullptr) {
397 word += delimiters.postfix;
398 }
399 return word;
400}

485inline std::string Pretty::underscore(const std::string& word) {
486 std::string str = std::regex_replace(word, std::regex("([A-Z]+)([A-Z][a-z])"), "$1_$2");
487 str = std::regex_replace(str, std::regex("([a-z\\d])([A-Z])"), "$1_$2");
488 str = std::regex_replace(str, std::regex("-"), "_");
489 std::ranges::transform(str, str.begin(), ::tolower);
490 return str;
491}

493// The last non-empty piece of s between delimiters, or an empty string when s has none.
494inline std::string Pretty::last(const std::string& s, const char delim) {
495 std::vector<std::string> elems;
496 std::stringstream ss(s);
497 std::string item;
498 while (getline(ss, item, delim)) {
499 if (!item.empty()) {
500 elems.push_back(item);
501 }
502 }
503 return elems.empty() ? std::string() : elems.back();
504}

517template <typename O>
518inline std::string Pretty::inspect_object(const O& o) {
519 if constexpr (Util::is_c_string<O>) {
520 // A C string is named by its type and printed quoted, or as nullptr when it is null.
521 constexpr std::string_view constness = std::is_const_v<std::remove_pointer_t<std::decay_t<O>>> ? "const " : "";
522 const std::string type = std::is_array_v<O> ? std::format("{}char [{}]", constness, std::extent_v<O>)
523 : std::format("{}char *", constness);
524 const auto text = string_view_of(o);
525 return text ? std::format("({}) => \"{}\"", type, *text) : std::format("({}) => nullptr", type);
526 } else {
527 return std::format("({}) => {}", Util::demangle(typeid(o).name()), to_word(o));
528 }
529}
