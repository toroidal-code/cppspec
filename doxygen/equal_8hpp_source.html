74// A null C string is a value of its own: it equals another null C string and differs from every piece of text.
75template <typename A, typename E>
76bool Equal<A, E>::match() {
77 if constexpr (compares_text) {
78 return string_view_of(this->actual()) == string_view_of(this->expected());
79 } else {
80 return this->expected() == this->actual();
81 }
82}

84template <typename A, typename E>
85std::string Equal<A, E>::comparison() {
86 if constexpr (compares_text) {
87 if (!string_view_of(this->actual()) || !string_view_of(this->expected())) {
88 return "A null C string equals only another null C string";
89 }
90 }
